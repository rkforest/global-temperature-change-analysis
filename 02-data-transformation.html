cols_to_keep <- c("Year", "J-D")

fn_annual <- function(df, id) {
  dft <- df |> 
    select(all_of(cols_to_keep)) |>
    rename(year = Year) |>
    rename(anomaly = `J-D`) |>
    mutate(geographic_id = id) |> 
    mutate(geographic_id = factor(geographic_id, levels=geographic_id_levels)) |> 
    mutate(year = as.integer(year)) |>
    mutate(decade = as.integer(year - (year %% 10) + 10)) |>
    mutate(climate_period = assign_climate_period(year)) |>
    select(geographic_id, climate_period, decade, year, anomaly) |> 
    drop_na()
  return(dft)
}

df1 <- fn_annual(df_raw_g, id="G")
df2 <- fn_annual(df_raw_n, id="N")
df3 <- fn_annual(df_raw_s, id="S")

df_annual <- bind_rows(df1, df2, df3)
df_annual <- df_annual |> filter(year >= analysis_start_year)
rm(df1, df2, df3)

  "Jan", "Feb", "Mar", "Apr", "May", "Jun",

fn_monthly <- function(df, id) {
  dft <- df |> 
    select(all_of(cols_to_keep)) |>
    pivot_longer(
      cols=c(Jan,Feb,Mar,Apr,May,Jun,Jul,Aug,Sep,Oct,Nov,Dec), 
      names_to = "month_abbr",
      values_to = "anomaly",
      values_drop_na = TRUE) |>
    rename(year = Year) |>
    mutate(month = match(month_abbr,month.abb)) |> 
    mutate(date = as.Date(sprintf("%04d-%02d-01",year,month))) |> 
    mutate(geographic_id = id) |> 
    mutate(geographic_id = factor(geographic_id, levels=geographic_id_levels)) |> 
    mutate(year = as.integer(year)) |>
    mutate(decade = as.integer(year - (year %% 10) + 10)) |>
    mutate(climate_period = assign_climate_period(year)) |>
    select(geographic_id, climate_period, decade, year, month, date, anomaly)

  return(dft)
}

Seasonal Annual

Meteorological seasons are hemisphere dependent:

  • NH winter, SH summer: December, January, February.
  • NH spring, SH autumn: March, April, May.
  • NH summer, SH winter: June, July, August.
  • NH autumn, SH spring: September, October, November.
Code 
cols_to_keep <- c("Year", "DJF", "MAM", "JJA", "SON")

fn_seasonal <- function(df, id) {
  dft <- df |> 
    select(all_of(cols_to_keep)) |>
    pivot_longer(
      cols=c(DJF, MAM, JJA, SON), 
      names_to = "months",
      values_to = "anomaly",
      values_drop_na = TRUE) |>
    rename(year = Year) |>
    mutate(geographic_id = id) |> 
    mutate(geographic_id = factor(geographic_id, geographic_id_levels)) |> 
    mutate(year = as.integer(year)) |>
    mutate(decade = as.integer(year - (year %% 10) + 10)) |>
    mutate(climate_period = assign_climate_period(year)) |>
    mutate(season = case_when(
      geographic_id == "N" & months == "DJF" ~ "Winter",
      geographic_id == "N" & months == "MAM" ~ "Spring",
      geographic_id == "N" & months == "JJA" ~ "Summer",
      geographic_id == "N" & months == "SON" ~ "Autumn",
      geographic_id == "S" & months == "DJF" ~ "Summer",
      geographic_id == "S" & months == "MAM" ~ "Autumn",
      geographic_id == "S" & months == "JJA" ~ "Winter",
      geographic_id == "S" & months == "SON" ~ "Spring"
    ))|>
    select(geographic_id, climate_period, decade, year, season, anomaly)
  return(dft)
}

df1 <- fn_seasonal(df_raw_n, id="N")
df2 <- fn_seasonal(df_raw_s, id="S")

df_seasonal <- bind_rows(df1, df2)

df_seasonal <- df_seasonal |> filter(year >= analysis_start_year)
rm(df1, df2)

fn_zonal <- function(df,select_cols,tidy_cols,id) {
  dft <- df |> 
    select(all_of(select_cols)) |>
    pivot_longer(
      cols =  all_of(tidy_cols),
      names_to = "zone",
      values_to = "anomaly",
      values_drop_na = TRUE) |> 
    rename(year = Year) |>
    mutate(zonal_id = id) |>
    mutate(zonal_id = factor(zonal_id)) |>
    mutate(zone = factor(zone, levels=zone_levels)) |>
    mutate(year = as.integer(year)) |>
    mutate(decade = as.integer(year - (year %% 10) + 10)) |> 
    mutate(climate_period = assign_climate_period(year)) |>
    select(zonal_id, zone, climate_period, year, anomaly)
  return(dft) 
}

tidy_cols <- colnames(df_raw_z[,5:7])
select_cols <- c("Year",tidy_cols)

period_levels <- climate_periods$climate_period

values_period <- simplify2array(
  lapply(
    period_levels,
    function(p) {

      period_index <-
        years >= analysis_start_year &
        !is.na(netCDF_period) &
        netCDF_period == p

      apply(
        values[
          ,
          ,
          period_index,
          drop = FALSE
        ],
        c(1, 2),
        mean,
        na.rm = TRUE
      )
    }
  )
)

Convert to spatial data frame

Create grid polygons
Code 
# Grid spacing
lon_resolution <- median(diff(sort(unique(lon))))
lat_resolution <- median(diff(sort(unique(lat))))

# Create one polygon for every netCDF grid cell
grid_geometry <- st_make_grid(
  st_as_sfc(
    st_bbox(
      c(
        xmin = min(lon) - lon_resolution / 2,
        xmax = max(lon) + lon_resolution / 2,
        ymin = min(lat) - lat_resolution / 2,
        ymax = max(lat) + lat_resolution / 2
      ),
      crs = st_crs(4326)
    )
  ),
  cellsize = c(
    lon_resolution,
    lat_resolution
  ),
  what = "polygons"
)

grid_centers <- expand.grid(
  lon = lon,
  lat = lat
)

grid_sf <- st_sf(
  cell_id = seq_len(nrow(grid_centers)),
  grid_centers,
  geometry = grid_geometry,
  crs = 4326
)

    transformation_datetime,

  )

```{r}
    mutate(decade = as.integer(year - (year %% 10) + 10)) |>
    mutate(climate_period = assign_climate_period(year)) |>
    select(geographic_id, climate_period, decade, year, month, date, anomaly)

  return(dft)
}
```

#### Seasonal Annual

Meteorological seasons are hemisphere dependent:  

  - NH winter, SH summer: December, January, February. 
  - NH spring, SH autumn: March, April, May. 
  - NH summer, SH winter: June, July, August. 
  - NH autumn, SH spring: September, October, November. 

```{r}
#| label: tabular-seasonal

cols_to_keep <- c("Year", "DJF", "MAM", "JJA", "SON")

fn_seasonal <- function(df, id) {
  dft <- df |> 
    select(all_of(cols_to_keep)) |>
    pivot_longer(
      cols=c(DJF, MAM, JJA, SON), 
      names_to = "months",
      values_to = "anomaly",
      values_drop_na = TRUE) |>
    rename(year = Year) |>
    mutate(geographic_id = id) |> 
    mutate(geographic_id = factor(geographic_id, geographic_id_levels)) |> 
    mutate(year = as.integer(year)) |>
    mutate(decade = as.integer(year - (year %% 10) + 10)) |>
    mutate(climate_period = assign_climate_period(year)) |>
    mutate(season = case_when(
      geographic_id == "N" & months == "DJF" ~ "Winter",
      geographic_id == "N" & months == "MAM" ~ "Spring",
      geographic_id == "N" & months == "JJA" ~ "Summer",
      geographic_id == "N" & months == "SON" ~ "Autumn",
      geographic_id == "S" & months == "DJF" ~ "Summer",
      geographic_id == "S" & months == "MAM" ~ "Autumn",
      geographic_id == "S" & months == "JJA" ~ "Winter",
      geographic_id == "S" & months == "SON" ~ "Spring"
    ))|>
    select(geographic_id, climate_period, decade, year, season, anomaly)
  return(dft)
}

df1 <- fn_seasonal(df_raw_n, id="N")
df2 <- fn_seasonal(df_raw_s, id="S")

df_seasonal <- bind_rows(df1, df2)

df_seasonal <- df_seasonal |> filter(year >= analysis_start_year)
rm(df1, df2)
```

```{r}
  st_as_sfc(
    st_bbox(
      c(
        xmin = min(lon) - lon_resolution / 2,
        xmax = max(lon) + lon_resolution / 2,
        ymin = min(lat) - lat_resolution / 2,
        ymax = max(lat) + lat_resolution / 2
      ),

)
```
