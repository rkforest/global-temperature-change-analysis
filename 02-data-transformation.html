assign_climate_period <- function(year) {
  cut(
    year,
    breaks = c(
      climate_periods$start_year,
      analysis_end_year + 1
    ),
    labels = climate_periods$period_abbr,
    right = FALSE,
    ordered_result = TRUE
  )
}

cols_to_keep <- c("Year", "J-D")

fn_annual <- function(df, id) {
  dft <- df |>
    select(all_of(cols_to_keep)) |>
    rename(year = Year) |>
    rename(anomaly = `J-D`) |>
    mutate(geographic_id = id) |>
    mutate(geographic_id = factor(geographic_id, levels = geographic_id_levels)) |>
    mutate(year = as.integer(year)) |>
    mutate(decade = as.integer(year - (year %% 10) + 10)) |>
    mutate(climate_period = assign_climate_period(year)) |>
    select(geographic_id, climate_period, decade, year, anomaly) |>
    drop_na()
}

df1 <- fn_annual(df_raw_g, id = "G")
df2 <- fn_annual(df_raw_n, id = "N")
df3 <- fn_annual(df_raw_s, id = "S")

df_annual <- df_annual |>

    year >= analysis_start_year,

fn_monthly <- function(df, id) {
  dft <- df |>
    select(all_of(cols_to_keep)) |>
    pivot_longer(
      cols = c(Jan, Feb, Mar, Apr, May, Jun, Jul, Aug, Sep, Oct, Nov, Dec),
      names_to = "month_abbr",
      values_to = "anomaly",
      values_drop_na = TRUE
    ) |>
    rename(year = Year) |>
    mutate(month = match(month_abbr, month.abb)) |>
    mutate(date = as.Date(sprintf("%04d-%02d-01", year, month))) |>
    mutate(geographic_id = id) |>
    mutate(geographic_id = factor(geographic_id, levels = geographic_id_levels)) |>
    mutate(year = as.integer(year)) |>
    mutate(decade = as.integer(year - (year %% 10) + 10)) |>
    mutate(climate_period = assign_climate_period(year)) |>
    select(geographic_id, climate_period, decade, year, month, date, anomaly)
}

cols_to_keep <- c("Year", "DJF", "MAM", "JJA", "SON")

fn_seasonal <- function(df, id) {
  dft <- df |>
    select(all_of(cols_to_keep)) |>
    pivot_longer(
      cols = c(DJF, MAM, JJA, SON),
      names_to = "months",
      values_to = "anomaly",
      values_drop_na = TRUE
    ) |>
    rename(year = Year) |>
    mutate(geographic_id = id) |>
    mutate(geographic_id = factor(geographic_id, geographic_id_levels)) |>
    mutate(year = as.integer(year)) |>
    mutate(decade = as.integer(year - (year %% 10) + 10)) |>
    mutate(climate_period = assign_climate_period(year)) |>
    mutate(season = case_when(
      geographic_id == "N" & months == "DJF" ~ "Winter",
      geographic_id == "N" & months == "MAM" ~ "Spring",
      geographic_id == "N" & months == "JJA" ~ "Summer",
      geographic_id == "N" & months == "SON" ~ "Autumn",
      geographic_id == "S" & months == "DJF" ~ "Summer",
      geographic_id == "S" & months == "MAM" ~ "Autumn",
      geographic_id == "S" & months == "JJA" ~ "Winter",
      geographic_id == "S" & months == "SON" ~ "Spring"
    )) |>
    select(geographic_id, climate_period, decade, year, season, anomaly)
}

fn_zonal <- function(df, select_cols, tidy_cols, id) {
  dft <- df |>
    select(all_of(select_cols)) |>
    pivot_longer(
      cols = all_of(tidy_cols),
      names_to = "zone",
      values_to = "anomaly",
      values_drop_na = TRUE
    ) |>
    rename(year = Year) |>
    mutate(zonal_id = id) |>
    mutate(zonal_id = factor(zonal_id)) |>
    mutate(zone = factor(zone, levels = zone_levels)) |>
    mutate(year = as.integer(year)) |>
    mutate(decade = as.integer(year - (year %% 10) + 10)) |>
    mutate(climate_period = assign_climate_period(year)) |>
    select(zonal_id, zone, climate_period, year, anomaly)
}

Show code 
values[values == fillvalue$value] <- NA

dates <- as.Date(
  time,
  origin = "1800-01-01"
)

years <- as.integer(
  format(dates, "%Y")
)

netcdf_period <- assign_climate_period(years)

period_levels <- climate_periods$period_abbr

values_period <- simplify2array(
  lapply(
    period_levels,
    function(p) {
      period_index <-
        years >= analysis_start_year &
          years <= analysis_end_year &
          !is.na(netcdf_period) &
          netcdf_period == p

      apply(
        values[
          , ,
          period_index,
          drop = FALSE
        ],
        c(1, 2),
        mean,
        na.rm = TRUE
      )
    }
  )
)

# Grid spacing
lon_resolution <- median(diff(sort(unique(lon))))
lat_resolution <- median(diff(sort(unique(lat))))

# Create one polygon for every netCDF grid cell
grid_geometry <- st_make_grid(
  st_as_sfc(
    st_bbox(
      c(
        xmin = min(lon) - lon_resolution / 2,
        xmax = max(lon) + lon_resolution / 2,
        ymin = min(lat) - lat_resolution / 2,
        ymax = max(lat) + lat_resolution / 2
      ),
      crs = st_crs(4326)
    )
  ),
  cellsize = c(
    lon_resolution,
    lat_resolution
  ),
  what = "polygons"
)

grid_centers <- expand.grid(
  lon = lon,
  lat = lat
)

grid_sf <- st_sf(
  cell_id = seq_len(nrow(grid_centers)),
  grid_centers,
  geometry = grid_geometry,
  crs = 4326
)

df_gridded <- expand.grid(
  cell_id = seq_len(length(lon) * length(lat)),
  climate_period = period_levels
) |>
  mutate(
    avg_anomaly = as.vector(values_period)
  ) |>
  filter(
    is.finite(avg_anomaly)
  ) |>
  mutate(
    climate_period = factor(
      climate_period,
      levels = period_levels,
      ordered = TRUE
    )
  )

sf_gridded <- df_gridded |>
  left_join(
    grid_sf,
    by = "cell_id"
  ) |>
  st_as_sf()

sf_gridded <- sf_gridded |>
  st_transform(mollweide_crs)

sf_gridded <- sf_gridded |>
  mutate(
    area_km2 = as.numeric(
      st_area(geometry)
    ) / 1e6
  )

  path = dir_ls(
    dir_transformed,
    type = "file"
  )
)

file_metadata <- file_info(transformed_catalog$path)

transformed_catalog <- transformed_catalog |>
  mutate(
    filename = path_file(path),
    format = path_ext(filename),
    directory = path_dir(path),
    transformation_datetime = format(
      file_metadata$modification_time,
      "%Y-%m-%d %H:%M"

      as.numeric(file_metadata$size) / 1024,
      0

    transformed_descriptions,

  ) |>
  arrange(filename) |>
  select(
    dataset,
    filename,
    format,
    directory,
    transformation_datetime,
    file_size_kb
  )

#| label: transform-gridded
values[values == fillvalue$value] <- NA

dates <- as.Date(
  time,
  origin = "1800-01-01"
)

years <- as.integer(
  format(dates, "%Y")
)

netcdf_period <- assign_climate_period(years)

period_levels <- climate_periods$period_abbr

values_period <- simplify2array(
  lapply(
    period_levels,
    function(p) {
      period_index <-
        years >= analysis_start_year &
          years <= analysis_end_year &
          !is.na(netcdf_period) &
          netcdf_period == p

      apply(
        values[
          , ,
          period_index,
          drop = FALSE
        ],
        c(1, 2),
        mean,
        na.rm = TRUE
      )
    }
  )
)

# Grid spacing
lon_resolution <- median(diff(sort(unique(lon))))
lat_resolution <- median(diff(sort(unique(lat))))

# Create one polygon for every netCDF grid cell
grid_geometry <- st_make_grid(
  st_as_sfc(
    st_bbox(
      c(
        xmin = min(lon) - lon_resolution / 2,
        xmax = max(lon) + lon_resolution / 2,
        ymin = min(lat) - lat_resolution / 2,
        ymax = max(lat) + lat_resolution / 2
      ),
      crs = st_crs(4326)
    )
  ),
  cellsize = c(
    lon_resolution,
    lat_resolution
  ),
  what = "polygons"
)

grid_centers <- expand.grid(
  lon = lon,
  lat = lat
)

grid_sf <- st_sf(
  cell_id = seq_len(nrow(grid_centers)),
  grid_centers,
  geometry = grid_geometry,
  crs = 4326
)

df_gridded <- expand.grid(
  cell_id = seq_len(length(lon) * length(lat)),
  climate_period = period_levels
) |>
  mutate(
    avg_anomaly = as.vector(values_period)
  ) |>
  filter(
    is.finite(avg_anomaly)
  ) |>
  mutate(
    climate_period = factor(
      climate_period,
      levels = period_levels,
      ordered = TRUE
    )
  )

sf_gridded <- df_gridded |>
  left_join(
    grid_sf,
    by = "cell_id"
  ) |>
  st_as_sf()

sf_gridded <- sf_gridded |>
  st_transform(mollweide_crs)

sf_gridded <- sf_gridded |>
  mutate(
    area_km2 = as.numeric(
      st_area(geometry)
    ) / 1e6
  )
