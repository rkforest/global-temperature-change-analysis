Download Temperature Anomalies

Show code 
for (i in seq_len(nrow(file_catalog))) {

  if (file_catalog$format[[i]] == "CSV") {

    filename <- file_catalog$filename[[i]]
    source_url <- file_catalog$source_url[[i]]

    download_path <- paste0(
      source_url,
      filename
    )

    latest_path <- path(dir_raw, filename)
    archive_path <- path(dir_archive, filename)

    download.file(
      url = download_path,
      destfile = latest_path,
      mode = "wb",
      quiet = TRUE
    )

    file_copy(
      path = latest_path,
      new_path = archive_path,
      overwrite = TRUE
    )

    message(
      "Updated: ",
      filename
    )
  }
}
Show code 
for (i in seq_len(nrow(file_catalog))) {
  if (file_catalog$format[[i]] == "NetCDF") {
    filename <- file_catalog$filename[[i]]
    source_url <- file_catalog$source_url[[i]]
    download_path <- paste0(
      source_url,
      filename,
      ".gz"
    )

    latest_path <- path(dir_raw, filename)
    archive_path <- path(dir_archive, paste0(filename,".gz"))

    temporary_path <- paste0(
      archive_path,
      ".download"
    )
    download.file(
      url = download_path,
      destfile = temporary_path,
      mode = "wb",
      quiet = FALSE
    )
    file_move(
      path = temporary_path,
      new_path = archive_path
    )
    R.utils::gunzip(
      filename = archive_path,
      destname = latest_path,
      overwrite = TRUE,
      remove = FALSE
    )
  }
}
