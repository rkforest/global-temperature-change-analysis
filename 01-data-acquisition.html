Download temperature anomalies

Download csv files

Code 
for (i in seq_len(nrow(file_catalog))) {

  if (file_catalog$format[[i]] == "CSV") {

    filename <- file_catalog$filename[[i]]
    source_url <- file_catalog$source_url[[i]]

    download_path <- paste0(
      source_url,
      filename
    )

    latest_path <- file.path(
      dir_raw,
      filename
    )

    archive_path <- file.path(
      dir_archive,
      filename
    )

    download.file(
      url = download_path,
      destfile = latest_path,
      mode = "wb",
      quiet = TRUE
    )

    file.copy(
      from = latest_path,
      to = archive_path,
      overwrite = TRUE
    )

    message(
      "Updated: ",
      filename
    )
  }
}

Download netCDF file

Code 
for (i in seq_len(nrow(file_catalog))) {
  if (file_catalog$format[[i]] == "NetCDF") {
    filename <- file_catalog$filename[[i]]
    source_url <- file_catalog$source_url[[i]]
    download_path <- paste0(
      source_url,
      filename,
      ".gz"
    )
    latest_path <- file.path(
      dir_raw,
      filename
    )
    archive_path <- file.path(
      dir_archive,
      paste0(
        filename,
        ".gz"
      )
    )
    temporary_path <- paste0(
      archive_path,
      ".download"
    )
    download.file(
      url = download_path,
      destfile = temporary_path,
      mode = "wb",
      quiet = FALSE
    )
    file.rename(
      from = temporary_path,
      to = archive_path
    )
    R.utils::gunzip(
      filename = archive_path,
      destname = latest_path,
      overwrite = TRUE,
      remove = FALSE
    )
  }
}

Acquire coastline

Code 
path_coastline <- file.path(dir_raw,"global-coastline.gpkg")
coastline_sf <- rnaturalearth::ne_download(
  scale = 10,
  type = "coastline",
  category = "physical",
  returnclass = "sf"
)
sf::st_write(
  coastline_sf,
  path_coastline,
  layer = "coastline",
  delete_dsn = TRUE,
  quiet = TRUE
)
file_catalog <-
  bind_rows(
    file_catalog,
    tibble(
      dataset = "Global Coastline",
      source = "Natural Earth",
      format = "GeoPackage",
      filename = basename(
        path_coastline
      ),
      source_url =
        "https://www.naturalearthdata.com/",
      local_path = path_coastline
    )
  )
